document.addEventListener("DOMContentLoaded", () => {
  const panels = document.querySelectorAll(".mobile-nav__panel");
  const triggers = document.querySelectorAll("[data-panel-trigger]");
  const backs = document.querySelectorAll("[data-panel-back]");
  const closeBtn = document.querySelector(".drawer__close-button");

  const activatePanel = (target) => {
    panels.forEach(panel => {
      panel.classList.remove("is-active");
    });
    const next = document.querySelector(`[data-panel='${target}']`);
    if (next) next.classList.add("is-active");
  };

  triggers.forEach(btn => {
    btn.addEventListener("click", () => {
      const target = btn.getAttribute("data-panel-trigger");
      activatePanel(target);
    });
  });

  backs.forEach(btn => {
    btn.addEventListener("click", () => {
      const target = btn.getAttribute("data-panel-back");
      activatePanel(target);
    });
  });

  // ❗ Reset nav to main menu on close
  if (closeBtn) {
    closeBtn.addEventListener("click", () => {
      activatePanel("main");  // This will reset the nav to the main menu
    });
  }
});

document.addEventListener("DOMContentLoaded", function () {
  document.querySelectorAll('.p2_quicklinks').forEach(link => {
    link.textContent = link.textContent

document.addEventListener("DOMContentLoaded", () => {
  const panels = document.querySelectorAll(".mobile-nav__panel");
  const triggers = document.querySelectorAll("[data-panel-trigger]");
  const backs = document.querySelectorAll("[data-panel-back]");
  const closeBtn = document.querySelector(".drawer__close-button");

  const activatePanel = (target) => {
    panels.forEach(panel => {
      panel.classList.remove("is-active");
    });
    const next = document.querySelector(`[data-panel='${target}']`);
    if (next) next.classList.add("is-active");
  };

  triggers.forEach(btn => {
    btn.addEventListener("click", () => {
      const target = btn.getAttribute("data-panel-trigger");
      activatePanel(target);
    });
  });

  backs.forEach(btn => {
    btn.addEventListener("click", () => {
      const target = btn.getAttribute("data-panel-back");
      activatePanel(target);
    });
  });

  // ❗ Reset nav to main menu on close
  if (closeBtn) {
    closeBtn.addEventListener("click", () => {
      activatePanel("main");  // This will reset the nav to the main menu
    });
  }
});

document.addEventListener("DOMContentLoaded", () => {
  const panels = document.querySelectorAll(".mobile-nav__panel");
  const triggers = document.querySelectorAll("[data-panel-trigger]");
  const backs = document.querySelectorAll("[data-panel-back]");
  const closeBtn = document.querySelector(".drawer__close-button");

  const activatePanel = (target) => {
    panels.forEach(panel => {
      panel.classList.remove("is-active");
    });
    const next = document.querySelector(`[data-panel='${target}']`);
    if (next) next.classList.add("is-active");
  };

  triggers.forEach(btn => {
    btn.addEventListener("click", () => {
      const target = btn.getAttribute("data-panel-trigger");
      activatePanel(target);
    });
  });

  backs.forEach(btn => {
    btn.addEventListener("click", () => {
      const target = btn.getAttribute("data-panel-back");
      activatePanel(target);
    });
  });

  // ❗ Reset nav to main menu on close
  if (closeBtn) {
    closeBtn.addEventListener("click", () => {
      activatePanel("main");  // This will reset the nav to the main menu
    });
  }
});

document.addEventListener('DOMContentLoaded', function () {
  const parentItems = document.querySelectorAll('.parent-item');
  const childLists = document.querySelectorAll('.child-list');

  function hideAllChildLists() {
    childLists.forEach(list => {
      list.style.display = 'none';
    });
  }

  parentItems.forEach((item) => {
    const key = item.getAttribute('data-parent-key');
    item.addEventListener('mouseenter', () => {
      hideAllChildLists();
      const target = document.querySelector(`.child-list[data-child-key="${key}"]`);
      if (target) {
        target.style.display = 'block';
      }
    });
  });

  const dropdown = document.querySelector('.header__dropdown');
  if (dropdown) {
    dropdown.addEventListener('mouseleave', () => {
      hideAllChildLists();
    });
  }
});
